     1→/**
     2→ * Монтаж игровых модулей (вынесено из bootstrap.ts по правилу 01-CORE §6:
     3→ * файл превысил 300 строк при касании M04.1 — сборка ядра и монтаж модулей
     4→ * разные ответственности).
     5→ *
     6→ * Здесь — ЕДИНСТВЕННОЕ место, где модули знают друг о друге: колбэки
     7→ * app-слоя вайрятся через ModuleRegistry (П4: прямые импорты запрещены;
     8→ * прецеденты D33 applyUseEffect, D38 getSubject/applyRolls).
     9→ *
    10→ * [M04.1 · C4] каналы магазина: deliverItem → inventory.acquire (D23),
    11→ * probeCapacity → inventory.canAcquire (проба до оплаты), rollItem →
    12→ * modforge.rollItem (именованный стрим shop:issue:N), applyRolls →
    13→ * inventory.setRolls; saveId — детерминизм ротации (карточка §2).
    14→ */
    15→import type { GameDirector } from './game-director';
    16→import type { ModuleRegistry } from './module-registry';
    17→import type { ContentRegistry } from '@/kernel/content/content-registry';
    18→import type { SaveDirector } from '@/kernel/save/save-director';
    19→import type { WalletService } from '@/kernel/services/wallet/wallet-service';
    20→import type { BuffService } from '@/kernel/services/buff/buff-service';
    21→import { PlayerModule } from './modules/player/module';
    22→import { InventoryModule } from './modules/inventory/module';
    23→import { ModForgeModule } from './modules/modforge/module';
    24→import { ShopModule } from './modules/shop/module';
    25→import { CraftModule } from './modules/craft/module';
    26→import { ForgeModule } from './modules/forge/module';
    27→import { FarmModule } from './modules/farm/module';
    28→import { PenModule } from './modules/pen/module';
    29→import { FishingModule } from './modules/fishing/module';
    30→import { LakeModule } from './modules/lake/module';
    31→import { WorkersModule } from './modules/workers/module';
    32→import { MineModule } from './modules/mine/module';
    33→import { QuestsModule } from './modules/quests/module';
    34→import type { DeliverReceipt } from './modules/shop/purchase';
    35→import type { PlayerTable } from '@/content/player/progression';
    36→import type { SkillDef } from '@/content/player/skills';
    37→import type { CounterDef } from '@/content/player/counters';
    38→import type { RolledModEntry } from '@/shared/types/rolled-mod';
    39→import type { AcquireReceipt } from './modules/inventory/store';
    40→import type { Result } from '@/shared/types/result';
    41→import type { CurrencyId, ID, QualityTier, StatId } from '@/shared/types/ids';
    42→import type { StationViewRow } from './modules/craft/stations';
    43→import { effectiveQualityOf } from './modules/forge/outcome';
    44→import type { ForgeTable } from '@/content/balance/forging';
    45→
    46→export function wireGameModules(
    47→  director: GameDirector,
    48→  modules: ModuleRegistry,
    49→  content: ContentRegistry,
    50→  save: SaveDirector,
    51→  wallet: WalletService,
    52→  // [M13.1] Buff-канал ядра + часы: швы mine (зелья крота — kernel.buff,
    53→  // BUF-1; expiresAt — абсолютной датой от now)
    54→  buffs: BuffService,
    55→  now: () => number,
    56→): void {
    57→  // --- M01 «Игрок»: активируется ДО init/load — срез и подписки должны
    58→  //     стоять до save.load() (иначе реген пропустит оффлайн-досчёт)
    59→  modules.register(
    60→    new PlayerModule(
    61→      content.get<PlayerTable>('player.progression'),
    62→      content.get<SkillDef[]>('player.skills'),
    63→      content.get<CounterDef[]>('player.counters'),
    64→    ),
    65→  );
    66→  director.activate('player');
    67→
    68→  // --- M02 «Инвентарь»: подмена ItemPort (D17) в activate; гейт-контекст и
    69→  //     канал эффектов useAction — через публичный API игрока (П4);
    70→  //     [M10.1 · шов §4.5] statScaleOf/effectiveTierOf — заточка в статах/цене
    71→  //     (forge через ModuleRegistry; до активации forge — дефолты 1/identity)
    72→  modules.register(
    73→    new InventoryModule({
    74→      gateExtras: () =>
    75→        (modules.call('player', 'player.ctxExtras') ?? {}) as Record<string, unknown>,
    76→      applyUseEffect: (t, value, reason) => {
    77→        if (t === 'grantEnergy') modules.call('player', 'player.grantEnergy', value, reason);
    78→        else modules.call('player', 'player.grantHp', value, reason);
    79→      },
    80→      statScaleOf: (iid) =>
    81→        (modules.call('forge', 'forge.statScale', iid) as number | undefined) ?? 1,
    82→      effectiveTierOf: (iid, tier) =>
    83→        (modules.call('forge', 'forge.effectiveTier', iid, tier) as QualityTier | undefined) ?? tier,
    84→    }),
    85→  );
    86→  director.activate('inventory');
    87→
    88→  // --- M03 «Модфордж»: доступ к предмету реролл-цикла — колбэки app-слоя
    89→  //     (прецедент D33); [M04.1 · C6] saveId — стабильный сид реролла
    90→  modules.register(
    91→    new ModForgeModule({
    92→      getSubject: (iid) => {
    93→        const items = (modules.call('inventory', 'inventory.list') ?? []) as Array<{
    94→          iid: string;
    95→          defId: string;
    96→          rolledMods: RolledModEntry[];
    97→        }>;
    98→        const it = items.find((e) => e.iid === iid);
    99→        return it ? { defId: it.defId, rolls: it.rolledMods } : null;
   100→      },
   101→      applyRolls: (iid, rolls) =>
   102→        (modules.call('inventory', 'inventory.setRolls', iid, rolls) ?? {
   103→          ok: false as const,
   104→          error: 'модуль инвентаря недоступен',
   105→        }) as Result<unknown>,
   106→      saveId: () => save.getSaveId(),
   107→    }),
   108→  );
   109→  director.activate('modforge');
   110→
   111→  // --- M04 «Магазин» (фаза 2): выдача/роллы/гейты — колбэки app-слоя (П4);
   112→  //     itemCount — по рюкзаку (inventory.list; equipped-записи гейтами
   113→  //     покупки не учитываются — владение уже состоявшейся выдачей)
   114→  //     [M15.2 · N10] gateExtras: гейты ассортимента читают флаги/ачивки M15
   115→  //     (unlock на строке пула/ItemDef → Requirement-оценщик → hasFlag/
   116→  //     hasAchievement из extras) — шов app-слоя, логика M04 не тронута
   117→  modules.register(
   118→    new ShopModule({
   119→      deliverItem: (defId, qty, source) =>
   120→        (modules.call('inventory', 'inventory.acquire', defId, qty, source) ?? {
   121→          ok: false as const,
   122→          error: 'модуль инвентаря недоступен',
   123→        }) as Result<DeliverReceipt>,
   124→      probeCapacity: (defId, qty) =>
   125→        (modules.call('inventory', 'inventory.canAcquire', defId, qty) ?? {
   126→          ok: false as const,
   127→          error: 'модуль инвентаря недоступен',
   128→        }) as Result<{ fits: boolean; free: number; newEntries: number }>,
   129→      rollItem: (defId, streamKey) =>
   130→        (modules.call('modforge', 'modforge.rollItem', defId, streamKey) ?? {
   131→          ok: false as const,
   132→          error: 'модуль модфорджа недоступен',
   133→        }) as Result<{ rolls: RolledModEntry[] }>,
   134→      applyRolls: (iid, rolls) =>
   135→        (modules.call('inventory', 'inventory.setRolls', iid, rolls) ?? {
   136→          ok: false as const,
   137→          error: 'модуль инвентаря недоступен',
   138→        }) as Result<unknown>,
   139→      gateExtras: () =>
   140→        ({
   141→          ...(modules.call('player', 'player.ctxExtras') ?? {}),
   142→          ...(modules.call('quests', 'quests.gateExtras') ?? {}),
   143→        }) as Record<string, unknown>,
   144→      itemCount: (defId) => {
   145→        const items = (modules.call('inventory', 'inventory.list') ?? []) as Array<{
   146→          defId: string;
   147→          qty: number;
   148→        }>;
   149→        return items.filter((i) => i.defId === defId).reduce((s, i) => s + i.qty, 0);
   150→      },
   151→      saveId: () => save.getSaveId(),
   152→    }),
   153→  );
   154→  director.activate('shop');
   155→
   156→  // --- M09 «Крафт» (фаза 2): выдача/роллы/энергия/гейты/ремонт — колбэки
   157→  //     app-слоя (П4, прецеденты D33/D38/C4); walletGet — чтение для плана
   158→  //     входов (списание — только TransactionService)
   159→  modules.register(
   160→    new CraftModule({
   161→      listItems: () =>
   162→        (modules.call('inventory', 'inventory.list') ?? []) as Array<{
   163→          iid: ID;
   164→          defId: ID;
   165→          qty: number;
   166→          quality: { tier: QualityTier; value: number };
   167→          durability?: { current: number; max: number };
   168→        }>,
   169→      walletGet: (id) => wallet.get(id as CurrencyId),
   170→      deliverOutput: (defId, tier, value, qty, source) =>
   171→        (modules.call('inventory', 'inventory.add', defId, tier, value, qty, source) ?? {
   172→          ok: false as const,
   173→          error: 'модуль инвентаря недоступен',
   174→        }) as Result<AcquireReceipt>,
   175→      probeCapacity: (defId, qty) =>
   176→        (modules.call('inventory', 'inventory.canAcquire', defId, qty) ?? {
   177→          ok: false as const,
   178→          error: 'модуль инвентаря недоступен',
   179→        }) as Result<{ fits: boolean; free: number; newEntries: number }>,
   180→      rollItem: (defId, streamKey) =>
   181→        (modules.call('modforge', 'modforge.rollItem', defId, streamKey) ?? {
   182→          ok: false as const,
   183→          error: 'модуль модфорджа недоступен',
   184→        }) as Result<{ rolls: RolledModEntry[] }>,
   185→      applyRolls: (iid, rolls) =>
   186→        (modules.call('inventory', 'inventory.setRolls', iid, rolls) ?? {
   187→          ok: false as const,
   188→          error: 'модуль инвентаря недоступен',
   189→        }) as Result<unknown>,
   190→      gateExtras: () =>
   191→        (modules.call('player', 'player.ctxExtras') ?? {}) as Record<string, unknown>,
   192→      withEnergy: (stat: StatId, action: () => Result<unknown>, reason?: string) =>
   193→        (modules.call('player', 'player.withEnergy', stat, action, reason) ?? {
   194→          ok: false as const,
   195→          error: 'модуль игрока недоступен',
   196→        }) as Result<unknown>,
   197→      setDurability: (iid, current) =>
   198→        (modules.call('inventory', 'inventory.dur', iid, current) ?? {
   199→          ok: false as const,
   200→          error: 'модуль инвентаря недоступен',
   201→        }) as Result<{ iid: ID; current: number; max: number }>,
   202→    }),
   203→  );
   204→  director.activate('craft');
   205→
   206→  // --- M10 «Заточка» (фаза 2): рюкзак/витрина станций/гейты — колбэки
   207→  //     app-слоя (П4, прецедент D38/C4); наковальня — через api craft.stations
   208→  //     («вкладка ЗАТОЧКИ у наковальни» карточки M09 §2 — вызов, не импорт);
   209→  //     [M10.2 · §5] карточные каналы: modforge.rollFor (бюджет 4×L, теги
   210→  //     card:forge, target R4) + inventory.setRolls (append выбранной карточки)
   211→  modules.register(
   212→    new ForgeModule({
   213→      listItems: () =>
   214→        (modules.call('inventory', 'inventory.list') ?? []) as Array<{
   215→          iid: ID;
   216→          defId: ID;
   217→          quality: { tier: QualityTier; value: number };
   218→          rolledMods?: RolledModEntry[];
   219→        }>,
   220→      craftStationBuilt: (id) =>
   221→        ((modules.call('craft', 'craft.stations') ?? []) as StationViewRow[]).some(
   222→          (s) => s.id === id && s.built,
   223→        ),
   224→      gateExtras: () =>
   225→        (modules.call('player', 'player.ctxExtras') ?? {}) as Record<string, unknown>,
   226→      rollFor: (req) =>
   227→        (modules.call('modforge', 'modforge.rollFor', req) ?? {
   228→          ok: false as const,
   229→          error: 'модуль модфорджа недоступен',
   230→        }) as Result<{ rolls: RolledModEntry[]; detail: { budget: number; refund: number; spent: number } }>,
   231→      applyRolls: (iid, rolls) =>
   232→        (modules.call('inventory', 'inventory.setRolls', iid, rolls) ?? {
   233→          ok: false as const,
   234→          error: 'модуль инвентаря недоступен',
   235→        }) as Result<unknown>,
   236→    }),
   237→  );
   238→  director.activate('forge');
   239→
   240→  // --- M05 «Ферма» (фаза 3): энергия/семя/урожай/вместимость — колбэки
   241→  //     app-слоя (П4, паттерн craft — прецеденты D33/D38/C4); удобрение —
   242→  //     валюта кошелька, модуль проводит её сам через TransactionService
   243→  //     из контекста активации; гейты культур — player.ctxExtras
   244→  //     [M05.3 · G6] + quests.gateExtras (N10-прецедент магазина): гейт
   245→  //     солнечника читает ачивку «Агроном» (achievement:<id> из флагов M15)
   246→  modules.register(
   247→    new FarmModule({
   248→      gateExtras: () =>
   249→        ({
   250→          ...(modules.call('player', 'player.ctxExtras') ?? {}),
   251→          ...(modules.call('quests', 'quests.gateExtras') ?? {}),
   252→        }) as Record<string, unknown>,
   253→      withEnergy: (stat, action, reason) =>
   254→        (modules.call('player', 'player.withEnergy', stat, action, reason) ?? {
   255→          ok: false as const,
   256→          error: 'модуль игрока недоступен',
   257→        }) as Result<unknown>,
   258→      removeSeed: (defId, qty, source) =>
   259→        (modules.call('inventory', 'inventory.remove', defId, qty, source) ?? {
   260→          ok: false as const,
   261→          error: 'модуль инвентаря недоступен',
   262→        }) as Result<unknown>,
   263→      deliverCrop: (defId, tier, value, qty, source) =>
   264→        (modules.call('inventory', 'inventory.add', defId, tier, value, qty, source) ?? {
   265→          ok: false as const,
   266→          error: 'модуль инвентаря недоступен',
   267→        }) as Result<unknown>,
   268→      probeCapacity: (defId, qty) =>
   269→        (modules.call('inventory', 'inventory.canAcquire', defId, qty) ?? {
   270→          ok: false as const,
   271→          error: 'модуль инвентаря недоступен',
   272→        }) as Result<{ fits: boolean; free: number; newEntries: number }>,
   273→    }),
   274→  );
   275→  director.activate('farm');
   276→
   277→  // --- M06 «Загон» (фаза 3): энергия/выдача продукции/вместимость — колбэки
   278→  //     app-слоя (П4, паттерн фермы — прецеденты D33/D38/C4); золото покупки
   279→  //     и Корм — модуль проводит сам через TransactionService из контекста
   280→  //     активации; гейты животных — player.ctxExtras + quests.gateExtras
   281→  //     (N10-прецедент: гейт Свиньи читает ачивку «Грибник»)
   282→  modules.register(
   283→    new PenModule({
   284→      gateExtras: () =>
   285→        ({
   286→          ...(modules.call('player', 'player.ctxExtras') ?? {}),
   287→          ...(modules.call('quests', 'quests.gateExtras') ?? {}),
   288→        }) as Record<string, unknown>,
   289→      withEnergy: (stat, action, reason) =>
   290→        (modules.call('player', 'player.withEnergy', stat, action, reason) ?? {
   291→          ok: false as const,
   292→          error: 'модуль игрока недоступен',
   293→        }) as Result<unknown>,
   294→      deliverProduct: (defId, tier, value, qty, source) =>
   295→        (modules.call('inventory', 'inventory.add', defId, tier, value, qty, source) ?? {
   296→          ok: false as const,
   297→          error: 'модуль инвентаря недоступен',
   298→        }) as Result<unknown>,
   299→      probeCapacity: (defId, qty) =>
   300→        (modules.call('inventory', 'inventory.canAcquire', defId, qty) ?? {
   301→          ok: false as const,
   302→          error: 'модуль инвентаря недоступен',
   303→        }) as Result<{ fits: boolean; free: number; newEntries: number }>,
   304→    }),
   305→  );
   306→  director.activate('pen');
   307→
   308→  // --- M07 «Рыбалка» (фаза 3): энергия/выдача/экипировка/износ/роллы —
   309→  //     колбэки app-слоя (П4, паттерн pen — прецеденты D33/D38/C4); золото
   310→  //     покупок и расход наживки модуль проводит сам через TransactionService
   311→  //     из контекста активации; гейт-условия рыб — player.ctxExtras +
   312→  //     quests.gateExtras (N10-прецедент); событие fish.caught поднимает
   313→  //     XP/навык/счётчики таблицами M01 с фазы 1 — без вайринга
   314→  modules.register(
   315→    new FishingModule({
   316→      gateExtras: () =>
   317→        ({
   318→          ...(modules.call('player', 'player.ctxExtras') ?? {}),
   319→          ...(modules.call('quests', 'quests.gateExtras') ?? {}),
   320→        }) as Record<string, unknown>,
   321→      withEnergy: (stat, action, reason) =>
   322→        (modules.call('player', 'player.withEnergy', stat, action, reason) ?? {
   323→          ok: false as const,
   324→          error: 'модуль игрока недоступен',
   325→        }) as Result<unknown>,
   326→      equippedView: () =>
   327→        (modules.call('inventory', 'inventory.equipped') ?? []) as Array<{
   328→          slot: string;
   329→          slotName: string;
   330→          entry: { iid: ID; defId: ID; qty: number };
   331→        }>,
   332→      acquire: (defId, qty, source) =>
   333→        (modules.call('inventory', 'inventory.acquire', defId, qty, source) ?? {
   334→          ok: false as const,
   335→          error: 'модуль инвентаря недоступен',
   336→        }) as Result<{ quality: { tier: QualityTier; value: number }; iids: ID[] }>,
   337→      probeCapacity: (defId, qty) =>
   338→        (modules.call('inventory', 'inventory.canAcquire', defId, qty) ?? {
   339→          ok: false as const,
   340→          error: 'модуль инвентаря недоступен',
   341→        }) as Result<{ fits: boolean; free: number; newEntries: number }>,
   342→      unequip: (slot) =>
   343→        (modules.call('inventory', 'inventory.unequip', slot) ?? {
   344→          ok: false as const,
   345→          error: 'модуль инвентаря недоступен',
   346→        }) as Result<unknown>,
   347→      equip: (iid, slot) =>
   348→        (modules.call('inventory', 'inventory.equip', iid, slot) ?? {
   349→          ok: false as const,
   350→          error: 'модуль инвентаря недоступен',
   351→        }) as Result<unknown>,
   352→      findEntryByDef: (defId) => {
   353→        const items = (modules.call('inventory', 'inventory.list') ?? []) as Array<{
   354→          iid: ID;
   355→          defId: string;
   356→        }>;
   357→        return items.find((i) => i.defId === defId)?.iid ?? null;
   358→      },
   359→      wear: (iid, n) =>
   360→        (modules.call('inventory', 'inventory.wear', iid, n) ?? {
   361→          ok: false as const,
   362→          error: 'модуль инвентаря недоступен',
   363→        }) as Result<{ current: number; max: number; broken: boolean }>,
   364→      // [M07.2 · F13] расход карты при «Открыть сейчас» (штатный канал M02)
   365→      removeItem: (defId, qty, source) =>
   366→        (modules.call('inventory', 'inventory.remove', defId, qty, source) ?? {
   367→          ok: false as const,
   368→          error: 'модуль инвентаря недоступен',
   369→        }) as Result<unknown>,
   370→      // [M07.2 · F14] исход skillXp событий-мини (publicApi M01)
   371→      grantSkillXp: (skill, amount, source) => {
   372→        modules.call('player', 'skills.addXp', skill, amount, source);
   373→      },
   374→      rollItem: (defId, streamKey, target) =>
   375→        (modules.call('modforge', 'modforge.rollItem', defId, streamKey, target) ?? {
   376→          ok: false as const,
   377→          error: 'модуль модфорджа недоступен',
   378→        }) as Result<{ rolls: RolledModEntry[] }>,
   379→      applyRolls: (iid, rolls) =>
   380→        (modules.call('inventory', 'inventory.setRolls', iid, rolls) ?? {
   381→          ok: false as const,
   382→          error: 'модуль инвентаря недоступен',
   383→        }) as Result<unknown>,
   384→      itemCount: (defId) => {
   385→        const items = (modules.call('inventory', 'inventory.list') ?? []) as Array<{
   386→          defId: string;
   387→          qty: number;
   388→        }>;
   389→        return items.filter((i) => i.defId === defId).reduce((s, i) => s + i.qty, 0);
   390→      },
   391→    }),
   392→  );
   393→  director.activate('fishing');
   394→
   395→  // --- M08 «Озеро» (фаза 3): энергия/инвентарь — колбэки app-слоя (П4,
   396→  //     паттерн pen/fishing — прецеденты D33/D38/C4); Корм/золото/лекарство —
   397→  //     модуль проводит сам через TransactionService из контекста активации;
   398→  //     дефы рыб — чтение общих таблиц контента (без импорта fishing);
   399→  //     монтаж ПОСЛЕ fishing ДО quests (бриф F2, прецедент pen → fishing)
