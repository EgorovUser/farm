     1→/**
     2→ * Монтаж игровых модулей (вынесено из bootstrap.ts по правилу 01-CORE §6:
     3→ * файл превысил 300 строк при касании M04.1 — сборка ядра и монтаж модулей
     4→ * разные ответственности).
     5→ *
     6→ * Здесь — ЕДИНСТВЕННОЕ место, где модули знают друг о друге: колбэки
     7→ * app-слоя вайрятся через ModuleRegistry (П4: прямые импорты запрещены;
     8→ * прецеденты D33 applyUseEffect, D38 getSubject/applyRolls).
     9→ *
    10→ * [M04.1 · C4] каналы магазина: deliverItem → inventory.acquire (D23),
    11→ * probeCapacity → inventory.canAcquire (проба до оплаты), rollItem →
    12→ * modforge.rollItem (именованный стрим shop:issue:N), applyRolls →
    13→ * inventory.setRolls; saveId — детерминизм ротации (карточка §2).
    14→ */
    15→import type { GameDirector } from './game-director';
    16→import type { ModuleRegistry } from './module-registry';
    17→import type { ContentRegistry } from '@/kernel/content/content-registry';
    18→import type { SaveDirector } from '@/kernel/save/save-director';
    19→import type { WalletService } from '@/kernel/services/wallet/wallet-service';
    20→import { PlayerModule } from './modules/player/module';
    21→import { InventoryModule } from './modules/inventory/module';
    22→import { ModForgeModule } from './modules/modforge/module';
    23→import { ShopModule } from './modules/shop/module';
    24→import { CraftModule } from './modules/craft/module';
    25→import { ForgeModule } from './modules/forge/module';
    26→import { FarmModule } from './modules/farm/module';
    27→import { PenModule } from './modules/pen/module';
    28→import { FishingModule } from './modules/fishing/module';
    29→import { LakeModule } from './modules/lake/module';
    30→import { WorkersModule } from './modules/workers/module';
    31→import { MineModule } from './modules/mine/module';
    32→import { QuestsModule } from './modules/quests/module';
    33→import type { DeliverReceipt } from './modules/shop/purchase';
    34→import type { PlayerTable } from '@/content/player/progression';
    35→import type { SkillDef } from '@/content/player/skills';
    36→import type { CounterDef } from '@/content/player/counters';
    37→import type { RolledModEntry } from '@/shared/types/rolled-mod';
    38→import type { AcquireReceipt } from './modules/inventory/store';
    39→import type { Result } from '@/shared/types/result';
    40→import type { CurrencyId, ID, QualityTier, StatId } from '@/shared/types/ids';
    41→import type { StationViewRow } from './modules/craft/stations';
    42→
    43→export function wireGameModules(
    44→  director: GameDirector,
    45→  modules: ModuleRegistry,
    46→  content: ContentRegistry,
    47→  save: SaveDirector,
    48→  wallet: WalletService,
    49→): void {
    50→  // --- M01 «Игрок»: активируется ДО init/load — срез и подписки должны
    51→  //     стоять до save.load() (иначе реген пропустит оффлайн-досчёт)
    52→  modules.register(
    53→    new PlayerModule(
    54→      content.get<PlayerTable>('player.progression'),
    55→      content.get<SkillDef[]>('player.skills'),
    56→      content.get<CounterDef[]>('player.counters'),
    57→    ),
    58→  );
    59→  director.activate('player');
    60→
    61→  // --- M02 «Инвентарь»: подмена ItemPort (D17) в activate; гейт-контекст и
    62→  //     канал эффектов useAction — через публичный API игрока (П4);
    63→  //     [M10.1 · шов §4.5] statScaleOf/effectiveTierOf — заточка в статах/цене
    64→  //     (forge через ModuleRegistry; до активации forge — дефолты 1/identity)
    65→  modules.register(
    66→    new InventoryModule({
    67→      gateExtras: () =>
    68→        (modules.call('player', 'player.ctxExtras') ?? {}) as Record<string, unknown>,
    69→      applyUseEffect: (t, value, reason) => {
    70→        if (t === 'grantEnergy') modules.call('player', 'player.grantEnergy', value, reason);
    71→        else modules.call('player', 'player.grantHp', value, reason);
    72→      },
    73→      statScaleOf: (iid) =>
    74→        (modules.call('forge', 'forge.statScale', iid) as number | undefined) ?? 1,
    75→      effectiveTierOf: (iid, tier) =>
    76→        (modules.call('forge', 'forge.effectiveTier', iid, tier) as QualityTier | undefined) ?? tier,
    77→    }),
    78→  );
    79→  director.activate('inventory');
    80→
    81→  // --- M03 «Модфордж»: доступ к предмету реролл-цикла — колбэки app-слоя
    82→  //     (прецедент D33); [M04.1 · C6] saveId — стабильный сид реролла
    83→  modules.register(
    84→    new ModForgeModule({
    85→      getSubject: (iid) => {
    86→        const items = (modules.call('inventory', 'inventory.list') ?? []) as Array<{
    87→          iid: string;
    88→          defId: string;
    89→          rolledMods: RolledModEntry[];
    90→        }>;
    91→        const it = items.find((e) => e.iid === iid);
    92→        return it ? { defId: it.defId, rolls: it.rolledMods } : null;
    93→      },
    94→      applyRolls: (iid, rolls) =>
    95→        (modules.call('inventory', 'inventory.setRolls', iid, rolls) ?? {
    96→          ok: false as const,
    97→          error: 'модуль инвентаря недоступен',
    98→        }) as Result<unknown>,
    99→      saveId: () => save.getSaveId(),
   100→    }),
   101→  );
   102→  director.activate('modforge');
   103→
   104→  // --- M04 «Магазин» (фаза 2): выдача/роллы/гейты — колбэки app-слоя (П4);
   105→  //     itemCount — по рюкзаку (inventory.list; equipped-записи гейтами
   106→  //     покупки не учитываются — владение уже состоявшейся выдачей)
   107→  //     [M15.2 · N10] gateExtras: гейты ассортимента читают флаги/ачивки M15
   108→  //     (unlock на строке пула/ItemDef → Requirement-оценщик → hasFlag/
   109→  //     hasAchievement из extras) — шов app-слоя, логика M04 не тронута
   110→  modules.register(
   111→    new ShopModule({
   112→      deliverItem: (defId, qty, source) =>
   113→        (modules.call('inventory', 'inventory.acquire', defId, qty, source) ?? {
   114→          ok: false as const,
   115→          error: 'модуль инвентаря недоступен',
   116→        }) as Result<DeliverReceipt>,
   117→      probeCapacity: (defId, qty) =>
   118→        (modules.call('inventory', 'inventory.canAcquire', defId, qty) ?? {
   119→          ok: false as const,
   120→          error: 'модуль инвентаря недоступен',
   121→        }) as Result<{ fits: boolean; free: number; newEntries: number }>,
   122→      rollItem: (defId, streamKey) =>
   123→        (modules.call('modforge', 'modforge.rollItem', defId, streamKey) ?? {
   124→          ok: false as const,
   125→          error: 'модуль модфорджа недоступен',
   126→        }) as Result<{ rolls: RolledModEntry[] }>,
   127→      applyRolls: (iid, rolls) =>
   128→        (modules.call('inventory', 'inventory.setRolls', iid, rolls) ?? {
   129→          ok: false as const,
   130→          error: 'модуль инвентаря недоступен',
   131→        }) as Result<unknown>,
   132→      gateExtras: () =>
   133→        ({
   134→          ...(modules.call('player', 'player.ctxExtras') ?? {}),
   135→          ...(modules.call('quests', 'quests.gateExtras') ?? {}),
   136→        }) as Record<string, unknown>,
   137→      itemCount: (defId) => {
   138→        const items = (modules.call('inventory', 'inventory.list') ?? []) as Array<{
   139→          defId: string;
   140→          qty: number;
   141→        }>;
   142→        return items.filter((i) => i.defId === defId).reduce((s, i) => s + i.qty, 0);
   143→      },
   144→      saveId: () => save.getSaveId(),
   145→    }),
   146→  );
   147→  director.activate('shop');
   148→
   149→  // --- M09 «Крафт» (фаза 2): выдача/роллы/энергия/гейты/ремонт — колбэки
   150→  //     app-слоя (П4, прецеденты D33/D38/C4); walletGet — чтение для плана
   151→  //     входов (списание — только TransactionService)
   152→  modules.register(
   153→    new CraftModule({
   154→      listItems: () =>
   155→        (modules.call('inventory', 'inventory.list') ?? []) as Array<{
   156→          iid: ID;
   157→          defId: ID;
   158→          qty: number;
   159→          quality: { tier: QualityTier; value: number };
   160→          durability?: { current: number; max: number };
   161→        }>,
   162→      walletGet: (id) => wallet.get(id as CurrencyId),
   163→      deliverOutput: (defId, tier, value, qty, source) =>
   164→        (modules.call('inventory', 'inventory.add', defId, tier, value, qty, source) ?? {
   165→          ok: false as const,
   166→          error: 'модуль инвентаря недоступен',
   167→        }) as Result<AcquireReceipt>,
   168→      probeCapacity: (defId, qty) =>
   169→        (modules.call('inventory', 'inventory.canAcquire', defId, qty) ?? {
   170→          ok: false as const,
   171→          error: 'модуль инвентаря недоступен',
   172→        }) as Result<{ fits: boolean; free: number; newEntries: number }>,
   173→      rollItem: (defId, streamKey) =>
   174→        (modules.call('modforge', 'modforge.rollItem', defId, streamKey) ?? {
   175→          ok: false as const,
   176→          error: 'модуль модфорджа недоступен',
   177→        }) as Result<{ rolls: RolledModEntry[] }>,
   178→      applyRolls: (iid, rolls) =>
   179→        (modules.call('inventory', 'inventory.setRolls', iid, rolls) ?? {
   180→          ok: false as const,
   181→          error: 'модуль инвентаря недоступен',
   182→        }) as Result<unknown>,
   183→      gateExtras: () =>
   184→        (modules.call('player', 'player.ctxExtras') ?? {}) as Record<string, unknown>,
   185→      withEnergy: (stat: StatId, action: () => Result<unknown>, reason?: string) =>
   186→        (modules.call('player', 'player.withEnergy', stat, action, reason) ?? {
   187→          ok: false as const,
   188→          error: 'модуль игрока недоступен',
   189→        }) as Result<unknown>,
   190→      setDurability: (iid, current) =>
   191→        (modules.call('inventory', 'inventory.dur', iid, current) ?? {
   192→          ok: false as const,
   193→          error: 'модуль инвентаря недоступен',
   194→        }) as Result<{ iid: ID; current: number; max: number }>,
   195→    }),
   196→  );
   197→  director.activate('craft');
   198→
   199→  // --- M10 «Заточка» (фаза 2): рюкзак/витрина станций/гейты — колбэки
   200→  //     app-слоя (П4, прецедент D38/C4); наковальня — через api craft.stations
   201→  //     («вкладка ЗАТОЧКИ у наковальни» карточки M09 §2 — вызов, не импорт);
   202→  //     [M10.2 · §5] карточные каналы: modforge.rollFor (бюджет 4×L, теги
   203→  //     card:forge, target R4) + inventory.setRolls (append выбранной карточки)
   204→  modules.register(
   205→    new ForgeModule({
   206→      listItems: () =>
   207→        (modules.call('inventory', 'inventory.list') ?? []) as Array<{
   208→          iid: ID;
   209→          defId: ID;
   210→          quality: { tier: QualityTier; value: number };
   211→          rolledMods?: RolledModEntry[];
   212→        }>,
   213→      craftStationBuilt: (id) =>
   214→        ((modules.call('craft', 'craft.stations') ?? []) as StationViewRow[]).some(
   215→          (s) => s.id === id && s.built,
   216→        ),
   217→      gateExtras: () =>
   218→        (modules.call('player', 'player.ctxExtras') ?? {}) as Record<string, unknown>,
   219→      rollFor: (req) =>
   220→        (modules.call('modforge', 'modforge.rollFor', req) ?? {
   221→          ok: false as const,
   222→          error: 'модуль модфорджа недоступен',
   223→        }) as Result<{ rolls: RolledModEntry[]; detail: { budget: number; refund: number; spent: number } }>,
   224→      applyRolls: (iid, rolls) =>
   225→        (modules.call('inventory', 'inventory.setRolls', iid, rolls) ?? {
   226→          ok: false as const,
   227→          error: 'модуль инвентаря недоступен',
   228→        }) as Result<unknown>,
   229→    }),
   230→  );
   231→  director.activate('forge');
   232→
   233→  // --- M05 «Ферма» (фаза 3): энергия/семя/урожай/вместимость — колбэки
   234→  //     app-слоя (П4, паттерн craft — прецеденты D33/D38/C4); удобрение —
   235→  //     валюта кошелька, модуль проводит её сам через TransactionService
   236→  //     из контекста активации; гейты культур — player.ctxExtras
   237→  //     [M05.3 · G6] + quests.gateExtras (N10-прецедент магазина): гейт
   238→  //     солнечника читает ачивку «Агроном» (achievement:<id> из флагов M15)
   239→  modules.register(
   240→    new FarmModule({
   241→      gateExtras: () =>
   242→        ({
   243→          ...(modules.call('player', 'player.ctxExtras') ?? {}),
   244→          ...(modules.call('quests', 'quests.gateExtras') ?? {}),
   245→        }) as Record<string, unknown>,
   246→      withEnergy: (stat, action, reason) =>
   247→        (modules.call('player', 'player.withEnergy', stat, action, reason) ?? {
   248→          ok: false as const,
   249→          error: 'модуль игрока недоступен',
   250→        }) as Result<unknown>,
   251→      removeSeed: (defId, qty, source) =>
   252→        (modules.call('inventory', 'inventory.remove', defId, qty, source) ?? {
   253→          ok: false as const,
   254→          error: 'модуль инвентаря недоступен',
   255→        }) as Result<unknown>,
   256→      deliverCrop: (defId, tier, value, qty, source) =>
   257→        (modules.call('inventory', 'inventory.add', defId, tier, value, qty, source) ?? {
   258→          ok: false as const,
   259→          error: 'модуль инвентаря недоступен',
   260→        }) as Result<unknown>,
   261→      probeCapacity: (defId, qty) =>
   262→        (modules.call('inventory', 'inventory.canAcquire', defId, qty) ?? {
   263→          ok: false as const,
   264→          error: 'модуль инвентаря недоступен',
   265→        }) as Result<{ fits: boolean; free: number; newEntries: number }>,
   266→    }),
   267→  );
   268→  director.activate('farm');
   269→
   270→  // --- M06 «Загон» (фаза 3): энергия/выдача продукции/вместимость — колбэки
   271→  //     app-слоя (П4, паттерн фермы — прецеденты D33/D38/C4); золото покупки
   272→  //     и Корм — модуль проводит сам через TransactionService из контекста
   273→  //     активации; гейты животных — player.ctxExtras + quests.gateExtras
   274→  //     (N10-прецедент: гейт Свиньи читает ачивку «Грибник»)
   275→  modules.register(
   276→    new PenModule({
   277→      gateExtras: () =>
   278→        ({
   279→          ...(modules.call('player', 'player.ctxExtras') ?? {}),
   280→          ...(modules.call('quests', 'quests.gateExtras') ?? {}),
   281→        }) as Record<string, unknown>,
   282→      withEnergy: (stat, action, reason) =>
   283→        (modules.call('player', 'player.withEnergy', stat, action, reason) ?? {
   284→          ok: false as const,
   285→          error: 'модуль игрока недоступен',
   286→        }) as Result<unknown>,
   287→      deliverProduct: (defId, tier, value, qty, source) =>
   288→        (modules.call('inventory', 'inventory.add', defId, tier, value, qty, source) ?? {
   289→          ok: false as const,
   290→          error: 'модуль инвентаря недоступен',
   291→        }) as Result<unknown>,
   292→      probeCapacity: (defId, qty) =>
   293→        (modules.call('inventory', 'inventory.canAcquire', defId, qty) ?? {
   294→          ok: false as const,
   295→          error: 'модуль инвентаря недоступен',
   296→        }) as Result<{ fits: boolean; free: number; newEntries: number }>,
   297→    }),
   298→  );
   299→  director.activate('pen');
   300→
   301→  // --- M07 «Рыбалка» (фаза 3): энергия/выдача/экипировка/износ/роллы —
   302→  //     колбэки app-слоя (П4, паттерн pen — прецеденты D33/D38/C4); золото
   303→  //     покупок и расход наживки модуль проводит сам через TransactionService
   304→  //     из контекста активации; гейт-условия рыб — player.ctxExtras +
   305→  //     quests.gateExtras (N10-прецедент); событие fish.caught поднимает
   306→  //     XP/навык/счётчики таблицами M01 с фазы 1 — без вайринга
   307→  modules.register(
   308→    new FishingModule({
   309→      gateExtras: () =>
   310→        ({
   311→          ...(modules.call('player', 'player.ctxExtras') ?? {}),
   312→          ...(modules.call('quests', 'quests.gateExtras') ?? {}),
   313→        }) as Record<string, unknown>,
   314→      withEnergy: (stat, action, reason) =>
   315→        (modules.call('player', 'player.withEnergy', stat, action, reason) ?? {
   316→          ok: false as const,
   317→          error: 'модуль игрока недоступен',
   318→        }) as Result<unknown>,
   319→      equippedView: () =>
   320→        (modules.call('inventory', 'inventory.equipped') ?? []) as Array<{
   321→          slot: string;
   322→          slotName: string;
   323→          entry: { iid: ID; defId: ID; qty: number };
   324→        }>,
   325→      acquire: (defId, qty, source) =>
   326→        (modules.call('inventory', 'inventory.acquire', defId, qty, source) ?? {
   327→          ok: false as const,
   328→          error: 'модуль инвентаря недоступен',
   329→        }) as Result<{ quality: { tier: QualityTier; value: number }; iids: ID[] }>,
   330→      probeCapacity: (defId, qty) =>
   331→        (modules.call('inventory', 'inventory.canAcquire', defId, qty) ?? {
   332→          ok: false as const,
   333→          error: 'модуль инвентаря недоступен',
   334→        }) as Result<{ fits: boolean; free: number; newEntries: number }>,
   335→      unequip: (slot) =>
   336→        (modules.call('inventory', 'inventory.unequip', slot) ?? {
   337→          ok: false as const,
   338→          error: 'модуль инвентаря недоступен',
   339→        }) as Result<unknown>,
   340→      equip: (iid, slot) =>
   341→        (modules.call('inventory', 'inventory.equip', iid, slot) ?? {
   342→          ok: false as const,
   343→          error: 'модуль инвентаря недоступен',
   344→        }) as Result<unknown>,
   345→      findEntryByDef: (defId) => {
   346→        const items = (modules.call('inventory', 'inventory.list') ?? []) as Array<{
   347→          iid: ID;
   348→          defId: string;
   349→        }>;
   350→        return items.find((i) => i.defId === defId)?.iid ?? null;
   351→      },
   352→      wear: (iid, n) =>
   353→        (modules.call('inventory', 'inventory.wear', iid, n) ?? {
   354→          ok: false as const,
   355→          error: 'модуль инвентаря недоступен',
   356→        }) as Result<{ current: number; max: number; broken: boolean }>,
   357→      // [M07.2 · F13] расход карты при «Открыть сейчас» (штатный канал M02)
   358→      removeItem: (defId, qty, source) =>
   359→        (modules.call('inventory', 'inventory.remove', defId, qty, source) ?? {
   360→          ok: false as const,
   361→          error: 'модуль инвентаря недоступен',
   362→        }) as Result<unknown>,
   363→      // [M07.2 · F14] исход skillXp событий-мини (publicApi M01)
   364→      grantSkillXp: (skill, amount, source) => {
   365→        modules.call('player', 'skills.addXp', skill, amount, source);
   366→      },
   367→      rollItem: (defId, streamKey, target) =>
   368→        (modules.call('modforge', 'modforge.rollItem', defId, streamKey, target) ?? {
   369→          ok: false as const,
   370→          error: 'модуль модфорджа недоступен',
   371→        }) as Result<{ rolls: RolledModEntry[] }>,
   372→      applyRolls: (iid, rolls) =>
   373→        (modules.call('inventory', 'inventory.setRolls', iid, rolls) ?? {
   374→          ok: false as const,
   375→          error: 'модуль инвентаря недоступен',
   376→        }) as Result<unknown>,
   377→      itemCount: (defId) => {
   378→        const items = (modules.call('inventory', 'inventory.list') ?? []) as Array<{
   379→          defId: string;
   380→          qty: number;
   381→        }>;
   382→        return items.filter((i) => i.defId === defId).reduce((s, i) => s + i.qty, 0);
   383→      },
   384→    }),
   385→  );
   386→  director.activate('fishing');
   387→
   388→  // --- M08 «Озеро» (фаза 3): энергия/инвентарь — колбэки app-слоя (П4,
   389→  //     паттерн pen/fishing — прецеденты D33/D38/C4); Корм/золото/лекарство —
   390→  //     модуль проводит сам через TransactionService из контекста активации;
   391→  //     дефы рыб — чтение общих таблиц контента (без импорта fishing);
   392→  //     монтаж ПОСЛЕ fishing ДО quests (бриф F2, прецедент pen → fishing)
   393→  modules.register(
   394→    new LakeModule({
   395→      withEnergy: (stat, action, reason) =>
   396→        (modules.call('player', 'player.withEnergy', stat, action, reason) ?? {
   397→          ok: false as const,
   398→          error: 'модуль игрока недоступен',
   399→        }) as Result<unknown>,
   400→      listInventory: () =>
   401→        (modules.call('inventory', 'inventory.list') ?? []) as Array<{
   402→          iid: ID;
